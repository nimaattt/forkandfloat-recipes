category: "Dessert"
tags:
  - "vegetarian"
  - "dessert"
prepTime: "15 mins"
cookTime: "45 mins"
servings: "8"
difficulty: "Easy"
coverImage: "https://lh3.googleusercontent.com/d/1tn0rvFsenglQ7WT2txc_ZQESK5VqRcb_"
date: 2026-09-21T23:59:24.679Z
emoji: "🍽️"
dietaryTags: ""
ingredients:
  - name: "Mild olive oil"
    amount: "90 ml, or 100 g melted butter"
  - name: "Sugar"
    amount: "80 g"
  - name: "Eggs"
    amount: "2"
  - name: "Plain Greek yogurt"
    amount: "170 g"
  - name: "Vanilla extract"
    amount: "1 tsp"
  - name: "All-purpose flour"
    amount: "200 g"
  - name: "Baking powder"
    amount: "1 1/2 tsp"
  - name: "Salt"
    amount: "a pinch"
  - name: "Peach"
    amount: "1 medium"
  - name: "Berries"
    amount: "150 g"
  - name: "Brown sugar"
    amount: "20 g"
  - name: "Flaked almonds"
    amount: "20 g"
  - name: "Cinnamon"
    amount: "1/4 tsp"
steps:
  - title: "Prepare the batter"
    body: "In a large bowl, whisk together the eggs, sugar, olive oil, Greek yogurt and vanilla until smooth and well combined."
  - title: "Add the dry ingredients"
    body: "Add the flour, baking powder and a pinch of salt. Fold everything together just until combined, being careful not to overmix."
  - title: "Add the fruit"
    body: "Gently fold the chopped peaches and strawberries as well as the blueberries into the batter until evenly distributed."
  - title: "Prepare the cake"
    body: "Pour the batter into a greased or parchment-lined 20 cm (8-inch) cake pan and spread the top evenly."
  - title: "Add the topping"
    body: "In a small bowl, combine the brown sugar, flaked almonds and cinnamon. Sprinkle evenly over the top of the cake."
  - title: "Bake"
    body: "Bake at 350°F (180°C) for 45–50 minutes, or until golden on top and a toothpick inserted into the centre comes out clean."
  - title: "Cool"
    body: "Allow the cake to cool in the pan for about 10–15 minutes before removing and slicing."
howToServe:
  - "Serve slightly warm on its own."
  - "Add a spoonful of whipped cream or a scoop on vanilla ice cream."
  - "Serve with fresh berries or stone fruit."
  - "Enjoy at room temperature with coffee or tea."
chefsNote: "Use a mild olive oil so it complements rather than overpowers the fruit. You can also replace the olive oil with 100 g melted butter.
For a gluten-free version, replace the 200 g all-purpose flour with 150 g almond flour and 50 g coconut flour."
