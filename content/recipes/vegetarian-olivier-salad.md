ingredients:
  - name: "Potatoes"
    amount: "4 medium"
  - name: "Eggs"
    amount: "5"
  - name: "Pickles"
    amount: "1 cup, diced"
  - name: "Cilantro"
    amount: "1/2 cup, chopped"
  - name: "Carrot"
    amount: "2/3 cup, grated"
  - name: "Mayonnaise"
    amount: "1/2 cup"
  - name: "Greek yogurt"
    amount: "2/3 cup"
  - name: "Dijon mustard"
    amount: "1 tbsp"
  - name: "Salt"
    amount: "to taste"
  - name: "Black pepper"
    amount: "to taste"
  - name: "Canned green peas"
    amount: "optional"
steps:
  - title: "Cook the Potatoes and Eggs"
    body: "Bring a medium pot of water to a boil. Add the potatoes and eggs and cook until both are fully cooked through. Remove from the water and allow them to cool completely."
  - title: "Prepare the Base"
    body: "Peel the potatoes and eggs. Transfer them to a large mixing bowl and mash until mostly smooth."
  - title: "Mix the Salad"
    body: "Add the diced pickles, grated carrot, and chopped cilantro to the bowl."
  - title: "Make the Dressing"
    body: "Add the mayonnaise, Greek yogurt, Dijon mustard, salt, and black pepper. Mix everything together until well combined and creamy."
  - title: "Chill"
    body: "Transfer the salad to the refrigerator and chill for at least 1 hour before serving."
howToServe:
  - "Serve cold as a side dish for BBQs and picnics."
  - "Enjoy on its own as a light lunch."
  - "Spread into a fresh baguette to make a classic Olivier salad sandwich."
  - "Garnish with extra chopped cilantro or sliced pickles if desired."
chefsNote: "Many versions of Olivier salad include green peas. Feel free to add a can of drained green peas for a more traditional variation. This salad tastes even better after a few hours in the fridge, making it a great make-ahead dish for gatherings and meal prep."
