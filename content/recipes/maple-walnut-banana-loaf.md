emoji: "🍽️"
dietaryTags: ""
ingredients:
  - name: "Overripe bananas"
    amount: "3"
  - name: "Vanilla extract"
    amount: "1 tsp"
  - name: "Egg"
    amount: "1"
  - name: "Milk of choice"
    amount: "1/2 cup"
  - name: "Maple syrup"
    amount: "1/2 cup"
  - name: "Butter"
    amount: "1/3 cup, melted"
  - name: "Almond flour"
    amount: "3/4 cup"
  - name: "All-purpose flour"
    amount: "2/3 cup"
  - name: "Baking powder"
    amount: "1 tbsp"
  - name: "Ground cinnamon"
    amount: "1 tbsp"
  - name: "Salt"
    amount: "1 pinch"
  - name: "Walnuts"
    amount: "1 cup"
  - name: "Ricotta cheese"
    amount: "1 cup, for icing, optional"
  - name: "Icing sugar"
    amount: "1 tbsp, for icing, optional"
steps:
  - title: "Preheat the oven"
    body: "Preheat the oven to 385°F (195°C). Chop the walnuts. Grease or line a loaf pan with parchment paper."
  - title: "Mix the wet ingredients"
    body: "Mash the bananas in a large bowl. Add the egg, milk, melted butter, vanilla extract, and maple syrup. Whisk until smooth."
  - title: "Add the dry ingredients"
    body: "Add the almond flour, all-purpose flour, baking powder, cinnamon, and a pinch of salt. Fold everything together until just combined. Gently fold in the chopped walnuts."
  - title: "Bake"
    body: "Pour the batter into the prepared loaf pan. Bake for about 35 minutes, or until a toothpick inserted into the center comes out clean."
  - title: "Make the icing"
    body: "Whip together the ricotta, icing sugar, and 1 tsp cinnamon until smooth and creamy."
  - title: "Cool and serve"
    body: "Allow the loaf to cool for at least 10–15 minutes before slicing. Serve as is, or top with the ricotta icing if desired."
howToServe:
  - "Serve warm with ricotta cheese"
  - "Enjoy with butter or your favorite nut butter"
  - "Pair with coffee or tea"
  - "Toast leftovers for breakfast"
chefsNote: "The riper your bananas are, the sweeter and more flavorful your loaf will be. 
This loaf freezes beautifully. Slice it before freezing so you can grab individual pieces whenever you need them."
